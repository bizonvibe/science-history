## Звуки (синтез, воспроизводимо)

```bash
# щелчок лупы о стекло
ffmpeg -f lavfi -i "sine=f=2400:d=0.25" -f lavfi -i "anoisesrc=c=white:a=0.6:d=0.25:seed=5" -filter_complex "[0]volume=0.35,afade=t=out:st=0.01:d=0.2:curve=exp[a];[1]highpass=f=1800,afade=t=out:st=0.004:d=0.05[b];[a][b]amix=inputs=2:normalize=0,aecho=0.6:0.3:18:0.25,loudnorm=I=-22:TP=-4" -ar 48000 -ac 2 loupe-click.wav
# шорох папируса
ffmpeg -f lavfi -i "anoisesrc=c=pink:a=0.5:d=1.0:seed=21" -filter_complex "[0]bandpass=f=2500:w=3000,tremolo=f=18:d=0.5,afade=t=in:d=0.35:curve=exp,afade=t=out:st=0.45:d=0.55,loudnorm=I=-30:TP=-8" -ar 48000 -ac 2 paper.wav
# музыка: подложка + тамбура, провал перед раскрытием (50,6–52 с), «Egyptian Discovery» с 52 с
ffmpeg -i neutral-drone.mp3 -i tambura-eb.mp3 -i egyptian-discovery.mp3 -filter_complex "[0]atrim=0:60,asetpts=N/SR/TB,volume=1.0[d];[1]atrim=20:80,asetpts=N/SR/TB,lowpass=f=2500,volume=0.28[t];[d][t]amix=inputs=2:normalize=0,afade=t=in:d=2,volume='if(between(t,50.6,52),1-0.75*(t-50.6)/1.4,if(between(t,52,56),0.25+0.75*(t-52)/4,1))':eval=frame,afade=t=out:st=57:d=3[bed];[2]adelay=52000|52000,volume=0.9[s];[bed][s]amix=inputs=2:normalize=0,loudnorm=I=-20:TP=-3" -ar 48000 -ac 2 -b:a 192k -t 60 hori-music.mp3
```
